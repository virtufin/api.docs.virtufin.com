    C# Client Library

    The Virtufin API Client library (Virtufin.Api.Client) provides a high-level C# interface for interacting with the Virtufin API gateway.

    Installation

    Reference the Virtufin.Api.Client project or NuGet package in your C# application.

    <ProjectReference Include="path/to/Virtufin.Api.Client/Virtufin.Api.Client.csproj" />

    Overview

    The client library provides three main classes:

    Class Description
    ApiClient Main entry point with async-only operations
    GatewayClient Low-level gateway operations (async-only)
    ServiceClient Service-specific method invocation (async-only)

    Note: This library is async-only. For synchronous usage patterns, wrap calls in Task.Run() or use a different approach.

    Quick Start

    using Virtufin.Api.Client;

// Create and dispose when done
using var client = new ApiClient("localhost", 5002);

// List available services
var services = client.ListServices();
Console.WriteLine($"Services: {string.Join(", ", services)}");

// Invoke a method
var response = await client.InvokeAsync("workmanager", "GetStatus", 
    new Dictionary<string, object?> { ["key"] = "value" });

    ApiClient

    The main client class providing both synchronous and asynchronous operations.

    Constructor

    public ApiClient(string host = "localhost", int grpcPort = 5002)

    Parameters: - host - The gateway host address (default: "localhost") - grpcPort - The gRPC port (default: 5002)

    Properties

    // The host address
public string Host { get; }

// The gRPC port
public int GrpcPort { get; }

// Direct access to GatewayClient
public GatewayClient Gateway { get; }

    Methods

    Service Discovery

    // All methods are async
Task<List<string>> ListServicesAsync()

Task<List<Dictionary<string, object?>>> ListMethodsAsync(string service)

    Example: 

    // Get all services
var services = await client.ListServicesAsync();

// Get methods for a service
var methods = await client.ListMethodsAsync("workmanager");
foreach (var method in methods)
{
    Console.WriteLine($"Method: {method["name"]}");
    Console.WriteLine($"  Input: {method["inputType"]}");
    Console.WriteLine($"  Output: {method["outputType"]}");
    Console.WriteLine($"  Streaming: {method["isServerStreaming"]}");
}

    Method Invocation

    // Async only
Task<Dictionary<string, object?>> InvokeAsync(string service, string method, Dictionary<string, object?>? requestData = null)

    Example: 

    // Simple invocation
var response = await client.InvokeAsync("workmanager", "GetStatus");

// With request data
var response = await client.InvokeAsync("workmanager", "CreateWorker",
    new Dictionary<string, object?>
    {
        ["name"] = "Worker 1",
        ["capacity"] = 10
    });

// Async version
var response = await client.InvokeAsync("workmanager", "GetWorker",
    new Dictionary<string, object?> { ["workerId"] = "123" });

    Dynamic Service Proxy

    The GatewayClient supports dynamic method invocation via DynamicObject:

    // Direct service access via dynamic proxy
var result = await client.Gateway.workmanager.GetStatus();
var workers = await client.Gateway.workmanager.ListWorkers(
    new Dictionary<string, object?> { ["status"] = "active" });

// Create a worker
var result = await client.Gateway.workmanager.CreateWorker(
    new Dictionary<string, object?>
    {
        ["name"] = "MyWorker",
        ["capacity"] = 10
    });

    This uses C# dynamic keyword with DynamicObject under the hood. Each property access on client.Gateway returns a ServiceClient that dynamically routes method calls through GatewayClient.InvokeAsync.

    Example with explicit service access: 

    // Get service explicitly
var wm = client.Gateway.GetService("workmanager");
var result = await wm.SomeMethod(requestData);

    State Management

    State operations are async-only:

    // Save state and broadcast event to subscribers
var response = await client.SaveStateAsync("state", "mykey", "myvalue");
Console.WriteLine($"Saved: {response.Success} - {response.Message}");

// Get single state value
var state = await client.GetStateAsync("state", "mykey");
Console.WriteLine($"Value: {state.Value}, Etag: {state.Etag}");

// Get all state for a service
var allState = await client.GetAllStateAsync("state");
foreach (var kvp in allState)
    Console.WriteLine($"{kvp.Key} = {kvp.Value}");

// Register keys explicitly for GetAllState
await client.RegisterKeysAsync("state", new[] { "key1", "key2" });

    Pub/Sub Operations

    Pub/Sub operations via GatewayClient:

    // Publish event (sync)
var pubResponse = client.Gateway.PublishEvent("my-topic", System.Text.Encoding.UTF8.GetBytes("hello"));

// Publish event async
await client.Gateway.PublishEventAsync("my-topic", System.Text.Encoding.UTF8.GetBytes("hello"));

// Subscribe to a topic (server streaming)
using var call = client.Gateway.SubscribeToTopic("my-topic");
while (await call.ResponseStream.MoveNext())
{
    var evt = call.ResponseStream.Current;
    Console.WriteLine($"Received: {evt.Topic} - {evt.Data}");
}

// Subscribe to Gateway events (services/topics/eventTypes)
using var call = client.Subscribe(services: new[] { "workmanager" }, topics: new[] { "updates" });
while (await call.ResponseStream.MoveNext())
{
    var evt = call.ResponseStream.Current;
    Console.WriteLine($"Event: {evt.Service}/{evt.Topic}");
}

    Streaming Subscriptions

    // Subscribe to events (server streaming)
AsyncServerStreamingCall<StreamEvent> Subscribe(
    IEnumerable<string>? services = null,
    IEnumerable<string>? topics = null,
    IEnumerable<string>? eventTypes = null,
    CancellationToken cancellationToken = default)

// Subscribe with handler callback (blocking)
Task SubscribeAsync(
    IStreamEventHandler handler,
    IEnumerable<string>? services = null,
    IEnumerable<string>? topics = null,
    IEnumerable<string>? eventTypes = null,
    CancellationToken cancellationToken = default)

// Watch system events (server streaming)
AsyncServerStreamingCall<SystemEvent> WatchSystemEvents(CancellationToken cancellationToken = default)

// Watch system events with handler callback (blocking)
Task WatchSystemEventsAsync(
    IStreamEventHandler handler,
    CancellationToken cancellationToken = default)

    Example: 

    // Subscribe to events with handler
var handler = new StreamEventHandler();
handler.EventReceived += e => Console.WriteLine($"Event: {e.Topic} - {e.EventType}");
handler.ErrorOccurred += ex => Console.WriteLine($"Error: {ex.Message}");

await client.SubscribeAsync(handler, topics: new[] { "my-topic" });

// Direct streaming call
using var call = client.Subscribe(services: new[] { "state" });
while (await call.ResponseStream.MoveNext())
{
    var evt = call.ResponseStream.Current;
    Console.WriteLine($"Received: {evt.Service}/{evt.Topic}");
}

    Resource Management

    void Close()
Task CloseAsync()
void Dispose()
ValueTask DisposeAsync()

    Example: 

    // Using block (preferred)
using (var client = new ApiClient())
{
    var services = client.ListServices();
}

// Or manual disposal
var client = new ApiClient();
try
{
    var services = client.ListServices();
}
finally
{
    client.Dispose();
}

    Service Client Factory

    ServiceClient GetService(string name)

    Returns a cached ServiceClient for the specified service.

    ServiceClient

    A convenience class for invoking methods on a specific service. Obtained via GatewayClient.GetService().

    Constructor

    internal ServiceClient(GatewayClient gateway, string serviceName)

    Methods

    // Synchronous
Dictionary<string, object?> Invoke(string method, Dictionary<string, object?>? requestData = null)

// Asynchronous
Task<Dictionary<string, object?>> InvokeAsync(string method, Dictionary<string, object?>? requestData = null)

    Example: 

    // Get service client
var workManager = client.Gateway.GetService("workmanager");

// Invoke methods on that service
var status = workManager.Invoke("GetStatus");
var worker = workManager.Invoke("GetWorker", new Dictionary<string, object?> { ["workerId"] = "123" });

// Async
var status = await workManager.InvokeAsync("GetStatus");

    Error Handling

    Invocation Errors

    When method invocation fails, Invoke throws InvalidOperationException:

    try
{
    var response = client.Invoke("workmanager", "UnknownMethod");
}
catch (InvalidOperationException ex)
{
    Console.WriteLine($"Invocation failed: {ex.Message}");
    // Example: "Gateway invoke failed: NOT_FOUND - Method 'UnknownMethod' not found"
}

    Service Not Found

    When a service is not found, GrpcChannelPool.GetChannel throws ArgumentException:

    try
{
    var client = new ApiClient();
    client.Gateway.GetChannel("nonexistent");
}
catch (ArgumentException ex)
{
    Console.WriteLine($"Service not found: {ex.Message}");
}

    Connection Errors

    Connection failures throw standard gRPC exceptions:

    try
{
    using var client = new ApiClient("invalid-host", 5002);
    var services = client.ListServices();
}
catch (Grpc.Core.RpcException ex)
{
    Console.WriteLine($"RPC failed: {ex.Status}");
}

    Complete Example

    using Virtufin.Api.Client;

class Program
{
    static async Task Main(string[] args)
    {
        Console.WriteLine("Virtufin API Client Demo");
        Console.WriteLine("========================\n");

        using var client = new ApiClient("localhost", 5002);

        // Discover services
        Console.WriteLine("Available Services:");
        var services = await client.ListServicesAsync();
        foreach (var service in services)
        {
            Console.WriteLine($"  - {service}");

            // List methods for each service
            var methods = await client.ListMethodsAsync(service);
            foreach (var method in methods.Take(3)) // First 3 methods
            {
                var streaming = method["isServerStreaming"] == "True" ? " (stream)" : "";
                Console.WriteLine($"    {method["name"]}{streaming}");
            }
            if (methods.Count > 3)
                Console.WriteLine($"    ... and {methods.Count - 3} more");
        }

        Console.WriteLine();

        // Invoke a method
        Console.WriteLine("Invoking workmanager.GetStatus:");
        try
        {
            var response = await client.InvokeAsync("workmanager", "GetStatus");
            Console.WriteLine($"Response: {System.Text.Json.JsonSerializer.Serialize(response)}");
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error: {ex.Message}");
        }
    }
}

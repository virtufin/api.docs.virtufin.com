    Deployment Guide

    Overview

    Virtufin can be deployed using Docker containers and Kubernetes. This guide covers both development and production deployment scenarios.

    Docker Deployment

    Backend (Virtufin.Api)

    Dockerfile

    FROM mcr.microsoft.com/dotnet/aspnet:9.0 AS base
WORKDIR /app
EXPOSE 8080

FROM mcr.microsoft.com/dotnet/sdk:9.0 AS build
WORKDIR /src
COPY ["Virtufin.Api/Virtufin.Api.csproj", "Virtufin.Api/"]
RUN dotnet restore "Virtufin.Api/Virtufin.Api.csproj"
COPY . .
WORKDIR "/src/Virtufin.Api"
RUN dotnet build "Virtufin.Api.csproj" -c Release -o /app/build

FROM build AS publish
RUN dotnet publish "Virtufin.Api.csproj" -c Release -o /app/publish

FROM base AS final
WORKDIR /app
COPY --from=publish /app/publish .
ENTRYPOINT ["dotnet", "Virtufin.Api.dll"]

    Build and Run

    # Build the image
docker build -t virtufin/api:latest -f src/Virtufin.Api/Dockerfile .

# Run with Dapr
dapr run --app-id virtufin-api \
    --app-port 8080 \
    --dapr-http-port 3500 \
    --dapr-grpc-port 50001 \
    -- dotnet run --project src/Virtufin.Api/Virtufin.Api.csproj

# Or run the container
docker run -p 5001:8080 \
    -e SERVICES_CONFIG_PATH=/config/services.yaml \
    -v /path/to/config:/config \
    virtufin/api:latest

    Frontend (VirtufinApp)

    Dockerfile

    FROM node:20-alpine AS build
WORKDIR /app
COPY package*.json ./
RUN npm ci
COPY . .
RUN npm run build

FROM nginx:alpine AS final
COPY --from=build /app/dist /usr/share/nginx/html
COPY nginx.conf /etc/nginx/conf.d/default.conf
EXPOSE 3000
CMD ["nginx", "-g", "daemon off;"]

    Build and Run

    # Build the image
docker build -t virtufin/frontend:latest -f src/VirtufinApp/Dockerfile .

# Run
docker run -p 3000:3000 virtufin/frontend:latest

    Docker Compose (Development)

    version: '3.8'

services:
  virtufin-api:
    build:
      context: .
      dockerfile: src/Virtufin.Api/Dockerfile
    ports:
      - "5001:8080"
      - "5002:8080"
      - "5003:8080"
    environment:
      - SERVICES_CONFIG_PATH=/config/services.yaml
    volumes:
      - ./config:/config
    depends_on:
      - dapr

  virtufin-frontend:
    build:
      context: .
      dockerfile: src/VirtufinApp/Dockerfile
    ports:
      - "3000:3000"
    depends_on:
      - virtufin-api

  dapr:
    image: daprio/dapr:latest
    command: "./daprd"
    ports:
      - "3500:3500"
      - "50001:50001"

    Kubernetes Deployment

    Prerequisites

    • Kubernetes 1.28+
    • Dapr Kubernetes mode enabled
    • kubectl configured

    Install Dapr

    # Initialize Dapr in Kubernetes
dapr init -k

# Verify Dapr is running
dapr status -k

    Deploy ConfigMap

    kubectl apply -f k8s/configmap.yaml

    Deploy API

    # k8s/api-deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: virtufin-api
  namespace: default
spec:
  replicas: 1
  selector:
    matchLabels:
      app: virtufin-api
  template:
    metadata:
      labels:
        app: virtufin-api
      annotations:
        dapr.io/enabled: "true"
        dapr.io/app-id: "virtufin-api"
        dapr.io/app-port: "8080"
    spec:
      containers:
        - name: api
          image: virtufin/api:latest
          imagePullPolicy: IfNotPresent
          ports:
            - containerPort: 8080
          env:
            - name: SERVICES_CONFIG_PATH
              value: /config/services.yaml
          volumeMounts:
            - name: config
              mountPath: /config
      volumes:
        - name: config
          configMap:
            name: virtufin-services-config
---
apiVersion: v1
kind: Service
metadata:
  name: virtufin-api
  namespace: default
spec:
  selector:
    app: virtufin-api
  ports:
    - port: 80
      targetPort: 8080
  type: ClusterIP

    kubectl apply -f k8s/api-deployment.yaml

    Deploy Frontend

    # k8s/frontend-deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: virtufin-frontend
  namespace: default
spec:
  replicas: 2
  selector:
    matchLabels:
      app: virtufin-frontend
  template:
    metadata:
      labels:
        app: virtufin-frontend
    spec:
      containers:
        - name: frontend
          image: virtufin/frontend:latest
          imagePullPolicy: IfNotPresent
          ports:
            - containerPort: 3000
---
apiVersion: v1
kind: Service
metadata:
  name: virtufin-frontend
  namespace: default
spec:
  selector:
    app: virtufin-frontend
  ports:
    - port: 80
      targetPort: 3000
  type: LoadBalancer

    kubectl apply -f k8s/frontend-deployment.yaml

    Verify Deployment

    # Check pods
kubectl get pods -l app=virtufin-api
kubectl get pods -l app=virtufin-frontend

# Check services
kubectl get svc virtufin-api
kubectl get svc virtufin-frontend

# View logs
kubectl logs -l app=virtufin-api

    Access the Application

    # Get frontend external IP
kubectl get svc virtufin-frontend

# Access the UI
# http://<external-ip>

    Nginx Configuration

    The frontend uses nginx with API proxy:

    server {
    listen 3000;
    server_name _;
    root /usr/share/nginx/html;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;
    }

    location /api/ {
        proxy_pass http://virtufin-api/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
    }
}

    Production Considerations

    Security

    1. TLS Termination: Configure TLS at ingress level
    2. CORS: Restrict CORS policies in production
    3. Authentication: Add authentication layer if needed
    4. Secrets: Use Kubernetes Secrets for sensitive data

    Scaling

    apiVersion: apps/v1
kind: Deployment
metadata:
  name: virtufin-api
spec:
  replicas: 3  # Scale API horizontally
  # ...

    Resource Limits

    resources:
  requests:
    memory: "256Mi"
    cpu: "200m"
  limits:
    memory: "512Mi"
    cpu: "500m"

    Health Checks

    livenessProbe:
  httpGet:
    path: /health
    port: 8080
  initialDelaySeconds: 10
  periodSeconds: 5

readinessProbe:
  httpGet:
    path: /healthz
    port: 8080
  initialDelaySeconds: 5
  periodSeconds: 5

    Ingress Configuration

    apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: virtufin-ingress
  annotations:
    nginx.ingress.kubernetes.io/rewrite-target: /
spec:
  rules:
    - host: virtufin.example.com
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: virtufin-frontend
                port:
                  number: 80
          - path: /api
            pathType: Prefix
            backend:
              service:
                name: virtufin-api
                port:
                  number: 80

    Troubleshooting

    Common Issues

    Pod Not Starting

    kubectl describe pod <pod-name>
kubectl logs <pod-name>

    Dapr Not Working

    # Check Dapr sidecar
kubectl get pods --show-labels | grep dapr

# View Dapr logs
kubectl logs <pod-name> -c daprd

    Cannot Connect to Backend Services

    1. Verify backend services are running
    2. Check DNS resolution: nslookup <service-name>
    3. Verify gRPC ports are accessible

    Debug Mode

    Enable detailed errors in configuration:

    builder.Services.AddGrpc(options =>
{
    options.EnableDetailedErrors = true;
});
